import { useMemo, useState } from 'react';

const apps = [
  { id: 'equipos', label: 'Equipes', icon: '▣', keys: ['WPFInstall7zip'] },
  { id: 'teamspeak3', label: 'TeamSpeak 3', icon: '▣', keys: ['WPFTweaksTelemetry'] },
  { id: 'teamspeak6', label: 'TeamSpeak 6', icon: '▣', keys: ['WPFFeatureNetFx3'] },
  { id: 'telegram', label: 'Telegram', icon: '▣', keys: ['WPFAppxTeams'] },
];

const defaultSelected = ['equipos', 'teamspeak3', 'telegram'];

const getInitialSelected = () => {
  if (typeof window === 'undefined') {
    return new Set(defaultSelected);
  }

  const params = new URLSearchParams(window.location.search);
  const selectedFromUrl = params.get('selected');

  if (!selectedFromUrl) {
    return new Set(defaultSelected);
  }

  const validIds = new Set(apps.map((app) => app.id));
  const ids = selectedFromUrl
    .split(',')
    .map((value) => value.trim())
    .filter((value) => validIds.has(value));

  return new Set(ids.length > 0 ? ids : defaultSelected);
};

const buildCommand = `& ([ScriptBlock]::Create((irm "https://github.com/ChrisTitusTech/winutil/releases/download/26.10.28/winutil.ps1"))) -Config "$env:USERPROFILE\\Downloads\\winutil-config.json"`;

export default function QuickConfigPage() {
  const [selected, setSelected] = useState(() => getInitialSelected());
  const [copied, setCopied] = useState(null);

  const selectedCount = selected.size;

  const json = useMemo(() => {
    const keys = [...new Set(
      [...selected].flatMap((id) => {
        const app = apps.find((item) => item.id === id);
        return app ? app.keys : [];
      })
    )];

    return JSON.stringify(keys, null, 2);
  }, [selected]);

  const toggle = (id) => {
    setSelected((current) => {
      const next = new Set(current);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });
  };

  const copy = async (value, label) => {
    try {
      await navigator.clipboard.writeText(value);
      setCopied(label);
      window.setTimeout(() => setCopied(null), 1500);
    } catch {
      setCopied(null);
    }
  };

  const buildShareLink = () => {
    if (typeof window === 'undefined') {
      return '';
    }

    const params = new URLSearchParams();
    if (selected.size > 0) {
      params.set('selected', [...selected].join(','));
    }

    const query = params.toString();
    return `${window.location.origin}${window.location.pathname}${query ? `?${query}` : ''}`;
  };

  const download = () => {
    const blob = new Blob([json], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const anchor = document.createElement('a');
    anchor.href = url;
    anchor.download = 'winutil-config.json';
    anchor.click();
    URL.revokeObjectURL(url);
  };

  return (
    <div
      style={{
        minHeight: '100vh',
        background: '#04080d',
        color: '#f2f5f8',
        fontFamily: 'Segoe UI, sans-serif',
        padding: '0 0 2rem',
      }}
    >
      <div
        style={{
          maxWidth: '1800px',
          margin: '0 auto',
          padding: '1.2rem 0.8rem 0',
        }}
      >
        <div
          style={{
            display: 'grid',
            gridTemplateColumns: 'repeat(4, minmax(0, 1fr))',
            gap: '0.75rem',
            background: '#090d12',
            border: '1px solid rgba(255,255,255,0.14)',
            borderRadius: '12px',
            overflow: 'hidden',
            padding: '0.65rem 0.7rem',
          }}
        >
          {apps.map((app) => {
            const active = selected.has(app.id);
            return (
              <button
                key={app.id}
                type="button"
                aria-pressed={active}
                onClick={() => toggle(app.id)}
                style={{
                  display: 'flex',
                  alignItems: 'center',
                  justifyContent: 'space-between',
                  gap: '0.8rem',
                  padding: '0.9rem 1rem',
                  borderRadius: '10px',
                  border: '1px solid rgba(255,255,255,0.12)',
                  background: active ? 'rgba(255,255,255,0.04)' : 'rgba(255,255,255,0.015)',
                  color: '#f9fbfc',
                  fontSize: '1.05rem',
                  fontWeight: 500,
                  cursor: 'pointer',
                  transition: 'all 0.15s ease',
                }}
              >
                <span>{app.label}</span>
                <span
                  aria-hidden="true"
                  style={{
                    width: '18px',
                    height: '18px',
                    display: 'inline-flex',
                    alignItems: 'center',
                    justifyContent: 'center',
                    border: '1px solid rgba(255,255,255,0.4)',
                    borderRadius: '4px',
                    fontSize: '0.72rem',
                    opacity: 0.9,
                  }}
                >
                  {app.icon}
                </span>
              </button>
            );
          })}
        </div>

        <div
          style={{
            marginTop: '2rem',
            background: 'rgba(255,255,255,0.012)',
            border: '1px solid rgba(255,255,255,0.12)',
            borderRadius: '16px',
            padding: '1.7rem 1.5rem 1.3rem',
          }}
        >
          <div
            style={{
              display: 'flex',
              justifyContent: 'space-between',
              alignItems: 'center',
              gap: '1rem',
              flexWrap: 'wrap',
            }}
          >
            <div style={{ fontSize: '1.1rem', color: '#f3f5f7', fontWeight: 500 }}>
              {selectedCount === 0
                ? '0 selecionado'
                : `${selectedCount} selecionado${selectedCount > 1 ? 's' : ''}`}
            </div>

            <div style={{ display: 'flex', gap: '0.8rem', flexWrap: 'wrap' }}>
              <button
                type="button"
                onClick={() => copy(json, 'json')}
                style={{
                  background: 'rgba(255,255,255,0.02)',
                  color: '#f3f6fa',
                  border: '1px solid rgba(255,255,255,0.18)',
                  borderRadius: '10px',
                  padding: '0.7rem 1.25rem',
                  fontSize: '1rem',
                  cursor: 'pointer',
                }}
              >
                {copied === 'json' ? 'Copiado' : 'Copiar JSON'}
              </button>

              <button
                type="button"
                onClick={() => copy(buildShareLink(), 'link')}
                style={{
                  background: 'rgba(255,255,255,0.02)',
                  color: '#f3f6fa',
                  border: '1px solid rgba(255,255,255,0.18)',
                  borderRadius: '10px',
                  padding: '0.7rem 1.25rem',
                  fontSize: '1rem',
                  cursor: 'pointer',
                }}
              >
                {copied === 'link' ? 'Copiado' : 'Compartilhar link'}
              </button>

              <button
                type="button"
                onClick={download}
                style={{
                  background: '#2b74ff',
                  color: '#fff',
                  border: 'none',
                  borderRadius: '10px',
                  padding: '0.85rem 1.5rem',
                  fontSize: '1rem',
                  cursor: 'pointer',
                  fontWeight: 700,
                }}
              >
                Baixar a configuração
              </button>
            </div>
          </div>

          <div
            style={{
              marginTop: '2.1rem',
              color: '#d7dde6',
              fontSize: '1.15rem',
              lineHeight: 1.6,
            }}
          >
            Selecione pelo menos um item para montar uma configuração.
          </div>

          <div
            style={{
              marginTop: '2rem',
              color: '#dfe6ee',
              fontSize: '1.12rem',
              lineHeight: 1.7,
            }}
          >
            Baixe a configuração e depois execute isso em uma janela elevada do PowerShell:
          </div>

          <div
            style={{
              marginTop: '1rem',
              display: 'flex',
              alignItems: 'center',
              justifyContent: 'space-between',
              gap: '0.8rem',
              background: '#0b1118',
              border: '1px solid rgba(255,255,255,0.12)',
              borderRadius: '12px',
              padding: '0.8rem 0.9rem',
              minHeight: '60px',
              overflow: 'hidden',
            }}
          >
            <code
              style={{
                color: '#eff4fb',
                fontSize: '0.97rem',
                overflow: 'hidden',
                textOverflow: 'ellipsis',
                whiteSpace: 'nowrap',
              }}
            >
              {buildCommand}
            </code>

            <button
              type="button"
              onClick={() => copy(buildCommand, 'command')}
              style={{
                background: 'rgba(255,255,255,0.08)',
                color: '#fff',
                border: '1px solid rgba(255,255,255,0.15)',
                borderRadius: '10px',
                padding: '0.7rem 1rem',
                fontSize: '0.98rem',
                cursor: 'pointer',
              }}
            >
              {copied === 'command' ? 'Copiado' : 'Cópia'}
            </button>
          </div>
        </div>
      </div>
    </div>
  );
}
